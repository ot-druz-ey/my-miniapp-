  <h2>AI Генератор Аваторок</h2>
  <p style="font-size: 14px; color: #aaa; text-align: center;">Опиши картинку на английском (например: cool neon cat)</p>

  <input type="text" id="promptInput" placeholder="Введите описание...">
  <button id="genBtn" onclick="generateImage()">Сгенерировать</button>

  <div id="resultContainer">
    <div id="loader" class="loader">Рисуем картинку... (~5 сек)</div>
    <img id="outputImg" alt="Сгенерированное изображение">
  </div>

  <script>
    // Инициализация Telegram WebApp
    const tg = window.Telegram.WebApp;
    tg.expand();

    function generateImage() {
      const prompt = document.getElementById('promptInput').value.trim();
      if (!prompt) {
        alert("Пожалуйста, введите описание!");
        return;
      }

      const btn = document.getElementById('genBtn');
      const loader = document.getElementById('loader');
      const img = document.getElementById('outputImg');

      btn.disabled = true;
      loader.style.display = 'block';
      img.style.display = 'none';

      // Используем бесплатный нейросетевой генератор Pollinations.ai
      const imageUrl = `https://pollinations.ai/p/${encodeURIComponent(prompt)}?width=512&height=512&seed=${Math.floor(Math.random() * 1000)}`;

      img.src = imageUrl;
      img.onload = () => {
        loader.style.display = 'none';
        img.style.display = 'block';
        btn.disabled = false;
      };
      img.onerror = () => {
        loader.style.display = 'none';
        alert("Ошибка генерации, попробуйте еще раз.");
        btn.disabled = false;
      };
    }
  </script>
